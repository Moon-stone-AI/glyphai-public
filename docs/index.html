import math
from dataclasses import dataclass
from typing import Dict

@dataclass
class GpuTelemetryProfile:
    host: str
    bdf: str
    model: str
    hours: float
    junction_temp_c: float
    hotspot_delta_c: float
    ecc_dbe: int
    ecc_contained: int
    sbe_rate_per_hr: float
    row_remap_events: int
    spare_rows_pct: float
    row_remap_fail: bool
    bus_lost: int
    nvlink_crc_rate: bool
    link_degraded: bool
    thermal_throttle: bool
    sdc_diag_fail: bool

def compute_gpu_health_score(g: GpuTelemetryProfile) -> Dict:
    penalties = {"memory": 0.0, "severe_xid": 0.0, "links": 0.0, "throttling": 0.0, "diagnostics": 0.0, "age": 0.0}
    verdicts = []
    
    # 1. Memory (max 35)
    mem = (g.ecc_contained * 3.0) + (g.ecc_dbe * 8.0) + (g.row_remap_events * 4.0)
    if g.sbe_rate_per_hr > 1.0: mem += 5.0
    if g.spare_rows_pct < 20: mem += 15.0
    penalties["memory"] = min(mem, 35.0)
    
    # 2. Severe Xid (max 25)
    penalties["severe_xid"] = min(g.bus_lost * 8.0, 25.0)
    
    # 3. Links (max 15)
    lnk = (5.0 if g.nvlink_crc_rate else 0.0) + (10.0 if g.link_degraded else 0.0)
    penalties["links"] = min(lnk, 15.0)
    
    # 4. Throttling (max 10)
    thr = (5.0 if g.thermal_throttle or g.junction_temp_c >= 88 else 0.0)
    if g.junction_temp_c >= 95: thr += 5.0
    penalties["throttling"] = min(thr, 10.0)
    
    # 5. Diagnostics (max 10)
    if g.sdc_diag_fail: penalties["diagnostics"] = 10.0
    
    # 6. Age (max 5, 1 pt per 8760 h)
    penalties["age"] = min(round((g.hours / 8760.0), 1), 5.0)
    
    score = 100.0 - sum(penalties.values())
    
    # Hard Caps
    if g.row_remap_fail:
        score = min(score, 10.0)
        verdicts.append("RMA: row-remapping capacity exhausted")
    if g.sdc_diag_fail:
        score = min(score, 10.0)
        verdicts.append("RMA: failed matrix-math Tensor Core diagnostic")
    if g.spare_rows_pct <= 0:
        score = min(score, 30.0)
        verdicts.append("0% spare rows left in memory banks")
        
    # Group Caps (uncorrectable memory incidents)
    uncorr = g.ecc_dbe + g.row_remap_events
    if uncorr >= 2:
        score = min(score, 49.0)
        verdicts.append("Repeated uncorrectable memory: not for sale")
    elif uncorr >= 1:
        score = min(score, 74.0)
        verdicts.append("Uncorrectable memory error: sell only with disclosure")
        
    score = max(0.0, round(score, 1))
    
    # Arrhenius physics acceleration: Ea = 0.7 eV
    T_base, T_j = 348.15, g.junction_temp_c + 273.15
    arrhenius_af = math.exp((0.7 / 8.61733e-5) * (1.0 / T_base - 1.0 / T_j))
    hazard = (1.45 / 80000.0) * math.pow(g.hours / 80000.0, 0.45) * max(0.2, arrhenius_af)
    if g.ecc_dbe > 0: hazard *= 3.5
    if g.hotspot_delta_c > 14: hazard *= 1.8
    
    survival_30d = round(math.exp(-hazard * 720.0) * 100.0, 1)
    
    return {
        "score": score,
        "band": "excellent" if score >= 90 else "good" if score >= 75 else "caution" if score >= 50 else "do_not_sell",
        "breakdown": penalties,
        "arrhenius_af": round(arrhenius_af, 2),
        "predicted_mtbf_hours": int(1.0 / hazard),
        "survival_30d_pct": survival_30d,
        "verdicts": verdicts
    }
